# 0.5.0-W1-232.1 設計與 RED 測試（companion）

本文件由 PM 起草，供執行者實作 `.claude/lib/test_command_parse.py` 並接線三個 hook。
驗收以票面 acceptance 為準，本文件是實作依據。

## 1. 設計決策

| 項目 | 決定 | 理由 |
|------|------|------|
| 切分方式 | 沿用 `lib/git_command_parse.parse_command_statements`，前綴略過沿用其 `_skip_prefix_wrapper_count` | 已處理 heredoc 剝離、換行轉分隔符、`2>&1`、引號語意；不在 lib 再寫第二套 tokenizer |
| 回傳形態 | `find_test_invocations(command) -> Optional[List[TestInvocation]]`，每筆帶 `runner` | 三個 hook 關心的執行器不同（環境檢查只關心 flutter／dart）；`None`／`[]` 失敗語意與 git_command_parse 一致 |
| pytest 納入 | 納入（`pytest`、`py.test`、`python[X.Y] [旗標] -m pytest`，可經 `uv run` 與包裝腳本啟動） | (1) 包裝腳本已定為唯一的逾時寫法，框架自身測試全以 pytest 執行，不納入則照規範執行的測試監控全看不到；(2) post-test 的成功輸出樣式已含 pytest 的 `N passed in Xs`，失敗評估本就為 pytest 設計；(3) `cli-error-feedback-hook`／`cli-failure-help-reminder-hook` 的抑制清單宣稱 pytest 已由測試 hook 處理，不納入則該宣稱為假 |
| 各 hook 關心的執行器 | pre-test-hook：flutter、dart；test-timeout-pre 與 post-test-hook：全部 | pre-test 做 Flutter SDK 與 pubspec 檢查，對 pytest／npm 無意義；開始時間寫入端與讀取端必須同一集合，否則時長落在上一次的開始時間 |
| 解析失敗（`None`） | 三個 hook 皆視為非測試 | 三者皆為提示／監控型，漏判只少一次檢查或計時，不擋命令 |
| post-test 的 ticket body 寫入豁免 | 保留不動 | 切分後 ticket 命令的引號參數已不會命中，豁免變為冗餘但無害；移除屬另一項評估，不在本票 |
| 不支援 | `bash -c '...'` | 命令位於引號字串內，需二次解析 |

## 2. 模組草稿（`.claude/lib/test_command_parse.py`）

執行者可調整命名與結構，但行為須通過第 3 節測試。模組 docstring 依 `git_command_parse.py` 慣例寫明用途、失敗語意、支援與不支援的前綴。

```python
import re
from dataclasses import dataclass
from pathlib import Path
from typing import List, Optional, Tuple

from .git_command_parse import _skip_prefix_wrapper_count, parse_command_statements

RUNNER_FLUTTER = "flutter"
RUNNER_DART = "dart"
RUNNER_NPM = "npm"
RUNNER_PYTEST = "pytest"
ALL_RUNNERS = frozenset({RUNNER_FLUTTER, RUNNER_DART, RUNNER_NPM, RUNNER_PYTEST})

TIMEOUT_WRAPPER_NAME = "run-with-timeout.py"
PYTEST_EXECUTABLES = frozenset({"pytest", "py.test"})
_PYTHON_EXECUTABLE_RE = re.compile(r"^python(\d+(\.\d+)?)?$")
_TIMEOUT_OPTIONS_WITH_VALUE = frozenset({"-s", "--signal", "-k", "--kill-after"})
_UV_OPTIONS_WITH_VALUE = frozenset({
    "--directory", "--project", "--python", "-p", "--with", "-w",
    "--with-editable", "--with-requirements", "--env-file", "--group",
    "--extra", "--package", "--index", "--index-url", "--extra-index-url",
    "--cache-dir", "--config-file",
})


@dataclass(frozen=True)
class TestInvocation:
    __test__ = False  # 避免 pytest 把 Test 開頭的類別當測試收集
    runner: str
    statement: Tuple[str, ...]


def _skip_options(tokens, with_value):
    index = 0
    while index < len(tokens) and tokens[index].startswith("-"):
        index += 2 if tokens[index] in with_value else 1
    return tokens[index:]


def _strip_timeout(tokens):            # timeout [選項] <時長> <命令>
    return _skip_options(tokens[1:], _TIMEOUT_OPTIONS_WITH_VALUE)[1:]


def _strip_uv_run(tokens):             # uv [全域選項] run [run 選項] <命令>；非 uv run 回 []
    rest = _skip_options(tokens[1:], _UV_OPTIONS_WITH_VALUE)
    if not rest or rest[0] != "run":
        return []
    return _skip_options(rest[1:], _UV_OPTIONS_WITH_VALUE)


def _strip_timeout_wrapper(tokens):    # [直譯器 [旗標]] run-with-timeout.py N -- <命令>；不是包裝呼叫回 None
    index = 0
    if _PYTHON_EXECUTABLE_RE.match(Path(tokens[0]).name):
        index = 1
        while index < len(tokens) and tokens[index].startswith("-"):
            index += 1
    if index >= len(tokens) or Path(tokens[index]).name != TIMEOUT_WRAPPER_NAME:
        return None
    rest = tokens[index + 1:]
    return rest[rest.index("--") + 1:] if "--" in rest else []


def _strip_one_launcher(tokens):
    skipped = _skip_prefix_wrapper_count(tokens)   # env／KEY=VAL／sudo／nohup／time／xargs
    if skipped:
        return tokens[skipped:]
    head = tokens[0]
    if head == "timeout":
        return _strip_timeout(tokens)
    if head == "fvm":
        return tokens[1:]
    if head == "uv":
        return _strip_uv_run(tokens)
    unwrapped = _strip_timeout_wrapper(tokens)
    return tokens if unwrapped is None else unwrapped


def _strip_launch_prefixes(tokens):    # 前綴可巢狀（包裝腳本內再接 uv run），反覆去除至穩定
    while tokens:
        stripped = _strip_one_launcher(tokens)
        if stripped == tokens:
            return tokens
        tokens = stripped
    return tokens


def _is_python_module_pytest(tokens):  # python [旗標] -m pytest
    if not _PYTHON_EXECUTABLE_RE.match(Path(tokens[0]).name):
        return False
    rest = tokens[1:]
    while rest and rest[0].startswith("-") and rest[0] != "-m":
        rest = rest[1:]
    return len(rest) >= 2 and rest[0] == "-m" and rest[1] == "pytest"


def _is_npm_test(tokens):              # npm test[:x] / npm run test[:x]
    if len(tokens) > 2 and tokens[1] == "run":
        script = tokens[2]
    else:
        script = tokens[1] if len(tokens) > 1 else ""
    return script == "test" or script.startswith("test:")


def _classify_runner(tokens):
    if not tokens:
        return None
    head = Path(tokens[0]).name
    if head in (RUNNER_FLUTTER, RUNNER_DART):
        return head if len(tokens) > 1 and tokens[1] == "test" else None
    if head == RUNNER_NPM:
        return RUNNER_NPM if _is_npm_test(tokens) else None
    if head in PYTEST_EXECUTABLES or _is_python_module_pytest(tokens):
        return RUNNER_PYTEST
    return None


def find_test_invocations(command: str) -> Optional[List[TestInvocation]]:
    statements = parse_command_statements(command)
    if statements is None:
        return None
    invocations = []
    for statement in statements:
        runner = _classify_runner(_strip_launch_prefixes(statement))
        if runner is not None:
            invocations.append(TestInvocation(runner=runner, statement=tuple(statement)))
    return invocations
```

## 3. RED 測試（`.claude/lib/tests/test_test_command_parse.py`）

匯入方式比照 `.claude/lib/tests/test_git_command_parse.py`（`sys.path.insert(0, <.claude>)` 後 `from lib.test_command_parse import ...`）。

```python
import pytest

from lib.test_command_parse import ALL_RUNNERS, find_test_invocations

WRAPPER = "python3 .claude/scripts/run-with-timeout.py 300 --"


def _runners(command):
    invocations = find_test_invocations(command)
    return None if invocations is None else [inv.runner for inv in invocations]


@pytest.mark.parametrize("command, expected", [
    ("flutter test", ["flutter"]),
    ("fvm flutter test test/unit", ["flutter"]),
    ("dart test", ["dart"]),
    ("FLUTTER_TEST_SEED=1 fvm flutter test", ["flutter"]),
    ("env A=1 flutter test", ["flutter"]),
    ("timeout 600 fvm flutter test", ["flutter"]),
    ("timeout -k 5 600 dart test", ["dart"]),
    ("cd app && fvm flutter test 2>&1 | tail -20", ["flutter"]),
    ("(cd packages/core && dart test) | tail -5", ["dart"]),
    ("cd app\nfvm flutter test", ["flutter"]),
    # 包裝腳本後接的命令（E 線轉交）
    (f"{WRAPPER} fvm flutter test", ["flutter"]),
    (".claude/scripts/run-with-timeout.py 60 -- flutter test", ["flutter"]),
    ("python3 -I .claude/scripts/run-with-timeout.py 600 -- fvm flutter test", ["flutter"]),
    ("uv run python3 .claude/scripts/run-with-timeout.py 600 -- dart test", ["dart"]),
    # pytest 三種形態
    ("pytest tests/test_x.py -q", ["pytest"]),
    ("python3 -m pytest tests/ -q", ["pytest"]),
    ("uv --directory .claude/hooks run --with pytest --with pyyaml python -m pytest tests/test_x.py -q", ["pytest"]),
    (f"{WRAPPER} uv --directory .claude/scripts run --with pytest python -m pytest tests -q", ["pytest"]),
    ("uv run pytest -q", ["pytest"]),
    # npm（既有 post-test 覆蓋範圍）
    ("npm test", ["npm"]),
    ("npm test:unit", ["npm"]),
    ("npm run test:hooks", ["npm"]),
    ("NODE_ENV=test npm test", ["npm"]),
    # 多個測試呼叫
    ("flutter test && pytest -q", ["flutter", "pytest"]),
])
def test_test_invocations_detected(command, expected):
    assert _runners(command) == expected


@pytest.mark.parametrize("command", [
    # 測試指令字樣在參數或引號內
    "grep -rn 'flutter test' .claude/rules",
    'grep -n "TEST_COMMAND_PATTERN\\|flutter test\\|dart test" hook.py',
    "echo x | grep 'dart test'",
    'ticket create --why "執行 flutter test 時 swap 耗盡" --what "閘門"',
    "ticket track append-log X --section Solution \"$(cat <<'EOF'\nfvm flutter test 全綠\nEOF\n)\"",
    'echo "npm test"',
    "echo dart test result",
    # 包裝器不在命令位置
    "grep run-with-timeout.py -- flutter test",
    # pytest 字樣只是 uv 選項值或其他命令的引數
    "uv run --with pytest python script.py",
    "pip install pytest",
    "python3 -m pip install pytest",
    # 非 test 子命令
    "flutter analyze",
    "npm run build",
    "uv sync",
    "",
])
def test_non_test_commands_not_detected(command):
    assert _runners(command) == []


def test_unparseable_command_returns_none():
    assert find_test_invocations("echo 'flutter test") is None


def test_statement_keeps_launch_prefix_for_display():
    (invocation,) = find_test_invocations(f"{WRAPPER} fvm flutter test")
    assert invocation.statement[0] == "python3"


def test_all_runners_constant_covers_every_runner():
    assert ALL_RUNNERS == {"flutter", "dart", "npm", "pytest"}
```

## 4. Hook 接線

| Hook | 改動 | 保留 |
