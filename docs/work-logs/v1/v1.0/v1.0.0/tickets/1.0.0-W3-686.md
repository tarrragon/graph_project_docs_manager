```python
import os, re, subprocess, sys
import yaml

PROJECT, FIXTURE = sys.argv[1], sys.argv[2]
P7 = "docs/proposals/PROP-007-in-app-editing.md"
S9 = "docs/spec/edit/SPEC-009-edit-write-path.md"
U1 = "docs/usecases/UC-01-open-project-and-reach-usable-state.md"
E1 = "docs/events/corpus/EVT-CORPUS-001-corpus-parsed.md"
CASES = [("PROP-007", P7, f, r) for f, r in [("target_version", 0), ("source", 0), ("priority", 0), ("evaluation_level", 0), ("proposed_by", 0), ("title", 1)]]
CASES += [("SPEC-009", S9, f, 0) for f in ["domain", "subdomain", "owner", "has_functional_requirements"]]
CASES += [("UC-01", U1, f, r) for f, r in [("primary_actor", 0), ("platform", 0), ("extension_status", 0), ("runtime_surface", 0), ("status", 1)]]
CASES += [("EVT-CORPUS-001", E1, f, r) for f, r in [("payload", 0), ("name", 1)]]
VARIANTS = {"baseline": None, "empty_string": '""', "null": "null", "remove_key": "<REMOVE>"}

def split_fm(text):
    m = re.match(r"^---\n(.*?\n)---\n", text, re.S)
    return m.group(1)

def mutate(text, field, repl):
    fm = split_fm(text); off = text.index(fm)
    m = re.compile(rf"^({re.escape(field)}:[ \t]*)([^\n#]*?)([ \t]*(#[^\n]*)?)$", re.M).search(fm)
    if not m: return None
    if repl == "<REMOVE>":
        return text[:off + m.start()] + text[off + m.end() + 1:]
    return text[:off + m.start(2)] + repl + text[off + m.end(2):]

def run(cmd):
    p = subprocess.run(cmd, cwd=PROJECT, env=dict(os.environ, CLAUDE_PROJECT_DIR=FIXTURE), capture_output=True, text=True)
    return p.returncode, (p.stdout + p.stderr).strip().splitlines()

def isolation(before, after, field):
    b, a = yaml.safe_load(split_fm(before)), yaml.safe_load(split_fm(after))
    added, removed = sorted(set(a) - set(b)), sorted(set(b) - set(a))
    changed = sorted(k for k in set(a) & set(b) if a[k] != b[k])
    ok = not added and not removed and set(changed) <= {field, "updated"}
    return f"parse_ok={ok} added={added} removed={removed} changed={changed} parsed={a.get(field, '<ABSENT>')!r}"

for doc_id, rel, field, req in CASES:
    path = os.path.join(FIXTURE, rel)
    original = open(path, encoding="utf-8").read()
    for vname, repl in VARIANTS.items():
        text = original if repl is None else mutate(original, field, repl)
        if text is None:
            print(f"{doc_id}\t{field}\t{vname}\tFIELD-NOT-FOUND"); continue
        open(path, "w", encoding="utf-8").write(text)
        rc_v, out_v = run(["doc", "validate", doc_id])
        rc_u, out_u = run(["doc", "uc", "verify", path])
        iso = "-" if repl is None else isolation(original, text, field)
        print(f"{doc_id}\t{field}\treq={req}\t{vname}\tvalidate_rc={rc_v}\tuc_rc={rc_u}\t{iso}")
        for line in out_v: print(f"    validate| {line}")
        for line in out_u[-2:]: print(f"    ucverify| {line}")
        open(path, "w", encoding="utf-8").write(original)
```

| 寫法 | 解析層隔離（FR-03 第 1 道近似） | `doc validate` | `doc uc verify` |
|---|---|---|---|
| 空字串 `""` | 全部通過（只有目標鍵的值改變） | 全部 rc=0，唯 EVT name rc=1「必填欄位值不可為空」 | 全部 rc=0 |

### 實驗發現

已驗證的事實：
1. `doc validate` 對 1.0 可編輯的 optional 欄位，三種寫法一律 rc=0，不區分空字串、null、缺鍵。PROP 沒有任何內容檢查；SPEC 只查 `depends_on_domains` 與 data-contract 章節；UC 只查 flow；EVT 只查 id/name/canonical_name/category 非空與 producers/consumers。驗證器不是這個問題的判別點。
2. 完整性集合欄位中，只有 EVT 的必填欄位會被攔下；PROP title、UC status 清空或移除鍵都 rc=0（驗證器缺口，見 Solution）。
3. 移除鍵使解析結果少一個鍵，違反 FR-03 允許變更集合（該條以「新增鍵會多一個鍵」為由不新增 `updated`，移除鍵同理）；位元組層也會改到值範圍以外的位元組。以現行 FR-03，移除鍵必回傳 `blocked(rewriteNotIsolated)`。
4. 空字串與 null 皆可在只換值範圍的前提下通過兩道隔離檢查。

仍未驗證的假設：
- App 端 `loadYamlNode` 對隱式 null（`key:` 後無值）的 span 長度；語料現行 null 皆為顯式 `null`，未遇到此形態。
